nums = [1, 2, 3, 4, 5, 6] # map：对每个元素做变换 squares = list(map(lambda x: x**2, nums)) # [1,4,9,16,25,36] # filter：筛选满足条件的 evens = list(filter(lambda x: x%2==0, nums)) # [2,4,6] # sorted：自定义排序规则 sorted_nums = sorted(nums, key=lambda x: -x) # 降序

📋 可变默认参数陷阱
Python 最经典的坑之一，面试必考
# ❌ 陷阱：默认参数只创建一次，共享！ def add_item(item, lst=[]): # [] 只创建一次 lst.append(item) return lst add_item(1) # [1] add_item(2) # [1, 2] ← 之前的还在！ # ✅ 正确做法：用 None def add_item(item, lst=None): if lst is None: lst = [] lst.append(item) return lst

Day 2 Python 进阶基础（下）
🎁 装饰器（@decorator）
在不改函数代码的前提下，给函数加新功能
# 装饰器：接收函数，返回新函数 def log(func): def wrapper(*args, **kwargs): print(f"调用 {func.__name__}") result = func(*args, **kwargs) print(f"调用结束") return result return wrapper @log # 等价于 say_hi = log(say_hi) def say_hi(name): print(f"你好，{name}") say_hi("小明") # 输出： # 调用 say_hi # 你好，小明 # 调用结束

📋 深浅拷贝
copy vs deepcopy，嵌套列表的坑
import copy a = [[1, 2], [3, 4]] b = a # 引用：改b就是改a c = a.copy() # 浅拷贝：外层独立，内层共享 d = copy.deepcopy(a) # 深拷贝：完全独立 a[0][0] = 99 print(b) # [[99,2],[3,4]] ← 变了 print(c) # [[99,2],[3,4]] ← 也变了！ print(d) # [[1,2],[3,4]] ← 没变 ✅

📊 实例方法 vs 类方法 vs 静态方法

class BankAccount: bank_name = "中国银行" # 类属性 def __init__(self, owner, balance): self.owner = owner # 实例属性 self.balance = balance # 实例方法：能访问 self（实例数据） def deposit(self, amount): self.balance += amount # 类方法：能访问 cls（类数据） @classmethod def get_bank_name(cls): return cls.bank_name # 静态方法：不能访问 self 也不能访问 cls @staticmethod def is_valid_amount(amount): return amount > 0

class Account: def __init__(self, owner, balance): self.owner = owner # 公开 self._type = "储蓄" # 约定私有 self.__balance = balance # 强制私有 # 通过方法控制访问（getter） def get_balance(self): return self.__balance # 通过方法控制修改（setter） def set_balance(self, value): if value < 0: print("余额不能为负数！") return self.__balance = value acc = Account("小明", 1000) print(acc.get_balance()) # 1000 acc.set_balance(-100) # 余额不能为负数！ acc.set_balance(2000) print(acc.get_balance()) # 2000

🏠 @property 装饰器
让方法像属性一样使用，不用写 get_xxx()
class Account: def __init__(self, balance): self.__balance = balance @property def balance(self): # 像属性一样读 return self.__balance @balance.setter def balance(self, value): # 像属性一样写 if value < 0: raise ValueError("不能为负") self.__balance = value acc = Account(1000) print(acc.balance) # 1000（不用写括号！） acc.balance = 2000 # 直接赋值（自动调 setter）

Day 5 继承与魔术方法
👨‍👦 继承和 super()
子类继承父类的属性和方法，可以复用代码
# 父类 class Animal: def __init__(self, name, age): self.name = name self.age = age def speak(self): print(f"{self.name}发出了声音") # 子类继承 Animal class Dog(Animal): def __init__(self, name, age, breed): super().__init__(name, age) # 调用父类的 __init__ self.breed = breed # 子类自己的属性 def speak(self): # 重写父类方法 print(f"{self.name}汪汪叫！") class Cat(Animal): def speak(self): print(f"{self.name}喵喵叫！") dog = Dog("旺财", 3, "金毛") dog.speak() # 旺财汪汪叫！ cat = Cat("咪咪", 2) cat.speak() # 咪咪喵喵叫！
核心知识点 实战：Animal → Dog/Cat
✨ 魔术方法（__str__、__eq__、__repr__）
让对象打印更好看、支持比较运算
class Student: def __init__(self, name, score): self.name = name self.score = score # 打印对象时显示什么（给人看） def __str__(self): return f"Student({self.name}, {self.score}分)" # 开发调试时显示什么（给程序员看） def __repr__(self): return f"Student(name='{self.name}', score={self.score})" # 支持 == 比较 def __eq__(self, other): return self.score == other.score # 支持 < 比较（用于排序） def __lt__(self, other): return self.score < other.score s1 = Student("小明", 90) s2 = Student("小红", 85) print(s1) # Student(小明, 90分) print(s1 == s2) # False print(s1 < s2) # False（90不小于85） # 排序 students = [s2, s1] students.sort() # 按 __lt__ 排序
方法触发时机
__str__print(对象) 或 str(对象)
__repr__直接显示对象、调试时
__eq__对象 == 对象
__lt__对象 < 对象
__len__len(对象)
__add__对象 + 对象

Day 6 迭代器、生成器、上下文管理
🔄 迭代器（__iter__、__next__）
让自定义类支持 for 循环遍历
class Counter: # 可迭代的：返回一个迭代器 def __iter__(self): self.n = 0 return self # 迭代器：返回下一个值 def __next__(self): self.n += 1 if self.n > 5: raise StopIteration # 结束信号 return self.n # 现在可以用 for 遍历了！ for num in Counter(): print(num) # 输出：1 2 3 4 5
for 循环的原理
⚡ 生成器（yield）
用 yield 创建迭代器，一次只产出一个值，省内存
# 生成器函数：遇到 yield 暂停，下次从暂停处继续 def counter(): n = 1 while n <= 5: yield n # 产出 n，暂停 n += 1 # 下次调用从这里继续 for num in counter(): print(num) # 输出：1 2 3 4 5 # 生成器表达式（类似列表推导式，但用圆括号） squares = (x**2 for x in range(1000000)) # 不占内存！一次只算一个
对比列表推导式 []生成器 ()
内存全部加载一次一个
能遍历几次多次一次
能索引✅❌
核心知识点 大数据处理利器
📦 上下文管理器（__enter__、__exit__）
让自定义类支持 with 语句，自动清理资源
class FileManager: def __init__(self, filename, mode): self.filename = filename self.mode = mode def __enter__(self): self.f = open(self.filename, self.mode, encoding="utf-8") return self.f # 返回给 as 后面的变量 def __exit__(self, exc_type, exc_val, exc_tb): self.f.close() # 自动关闭 return True # True 表示吞掉异常 # 现在可以用 with 了！ with FileManager("test.txt", "w") as f: f.write("hello") # 出了 with 块，自动调 __exit__ 关文件
with 语句的原理
